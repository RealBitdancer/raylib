function(add_raylib_regression name)
  add_executable(${name} ${name}.c)
  target_link_libraries(${name} PRIVATE raylib)
  set_target_properties(${name} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  add_test(NAME ${name} COMMAND ${name})
endfunction()

add_raylib_regression(core_file_regression)

if(SUPPORT_MODULE_RMODELS)
  add_raylib_regression(models_animation_regression)
endif()

if(SUPPORT_MODULE_RTEXTURES AND SUPPORT_IMAGE_GENERATION)
  add_raylib_regression(image_tint_regression)
endif()

  target_include_directories(raudio_regression PRIVATE ${PROJECT_SOURCE_DIR}/src)
  set_target_properties(raudio_regression PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  if(WIN32)
    target_link_libraries(raudio_regression PRIVATE winmm ole32)

    find_package(Threads REQUIRED)
    target_link_libraries(raudio_regression PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
  add_test(NAME raudio_regression COMMAND raudio_regression)
endif()
